  <section class="guide-sec" aria-labelledby="g-glance">
    <h2 id="g-glance">CLAT 2027 at a glance</h2>
    <dl class="facts">
      <div class="fact"><dt>Exam date</dt><dd class="fact-v">Sunday, 6 December 2026</dd><dd class="fact-n">2:00 p.m. to 4:00 p.m. Eligible PwD candidates until 4:40 p.m.</dd></div>
      <div class="fact"><dt>Applications</dt><dd class="fact-v">3 Aug to 31 Oct 2026</dd><dd class="fact-n">Close at 11:59 p.m. on 31 October. Online.</dd></div>
      <div class="fact"><dt>Application fee</dt><dd class="fact-v">₹4,000 · ₹3,500</dd><dd class="fact-n">₹4,000 for General and OBC. ₹3,500 for SC, ST, BPL and PwD. Non-refundable.</dd></div>
      <div class="fact"><dt>The paper</dt><dd class="fact-v">120 questions, 2 hours</dd><dd class="fact-n">120 marks. Five sections. The exam is held offline.</dd></div>
      <div class="fact"><dt>Marking</dt><dd class="fact-v">+1 · −0.25 · 0</dd><dd class="fact-n">+1 for a right answer, −0.25 for a wrong one, 0 if unanswered.</dd></div>
      <div class="fact"><dt>Eligibility</dt><dd class="fact-v">45% in Class 12</dd><dd class="fact-n">40% for SC, ST and PwD. No upper age limit. Class 12 students can apply.</dd></div>
    </dl>
  </section>
